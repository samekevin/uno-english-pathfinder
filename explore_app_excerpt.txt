  async function runAutoplay(token){
    autoplayRunning=true;
    overlay?.classList.add('is-idle');
    const candidates=autoplayCandidates();
    for(const node of candidates.slice(0,AUTOPLAY_CYCLES)){
      if(destroyed||token!==autoplayToken)break;
      focusNode(node.id,{autoplay:true});
      await wait(AUTOPLAY_HOLD_MS);
      if(destroyed||token!==autoplayToken)break;
      resetFocus({autoplay:true});
      await wait(AUTOPLAY_HOME_REST_MS);
    }
    autoplayRunning=false;
    overlay?.classList.remove('is-idle');
    lastActivity=Date.now();
    scheduleAutoplay();
  }

  function layoutProgramsNodes({nodes,width,height}){
    const base=layoutExploreNodes({nodes,centerId:'programs',width,height,rootMode:false});
    const p=base.get('programs');
    if(!p)return base;
    const mobile=width<620;
    const short=Math.min(width,height);
    const primaryRadius=short*(mobile?.235:.27);
    const secondaryRadius=short*(mobile?.34:.39);
    const tertiaryRadius=short*(mobile?.43:.50);
    // Four starting doors: BA / Graduate Studies / Minors / Certificates.
    const primaryAngles=[-155,-25,155,25].map(d=>d*Math.PI/180);
    programPrimaryIds.forEach((id,i)=>{
      const pos=base.get(id); if(!pos)return;
      const a=primaryAngles[i];
      pos.x=Math.cos(a)*primaryRadius;
      pos.y=Math.sin(a)*primaryRadius*.72;
      pos.layer=1;
      base.set(id,pos);
    });
    // Undergraduate pathways occupy the middle depth, loosely downstream of the BA.
    const secondaryAngles=[120,145,170,195,220].map(d=>d*Math.PI/180);
    programSecondaryIds.forEach((id,i)=>{
      const pos=base.get(id); if(!pos)return;
      const a=secondaryAngles[i];
      pos.x=Math.cos(a)*secondaryRadius;
      pos.y=Math.sin(a)*secondaryRadius*.70+short*.06;
      pos.layer=2;
      // Keep EXPLORE!'s typography language uniform; depth is carried mostly by position/opacity.
      pos.radius*=mobile?.94:.96;
      base.set(id,pos);
    });
    // Graduate certificates begin farther back, but can foreground normally when followed.
    const tertiaryAngles=[18,48,78,108,138].map(d=>d*Math.PI/180);
    programTertiaryIds.forEach((id,i)=>{
      const pos=base.get(id); if(!pos)return;
      const a=tertiaryAngles[i];
      pos.x=Math.cos(a)*tertiaryRadius;
      pos.y=Math.sin(a)*tertiaryRadius*.72+short*.08;
      pos.layer=3;
      pos.radius*=mobile?.90:.93;
      base.set(id,pos);
    });
    p.layer=0;
    base.set('programs',p);
    return base;
  }

  function stepMotion(ts){
    const ids=[...positions.keys()];
    if(reduced){
      for(const id of ids){const m=motionState.get(id);if(m){m.x=0;m.y=0;m.vx=0;m.vy=0;m.oscX=0;m.oscY=0;}}
      return;
    }
    const t=ts/1000;
    const width=view?.clientWidth||760;
    const height=view?.clientHeight||520;
    const coarse=window.matchMedia('(pointer:coarse)').matches;
    const {mobileProfile,motionScale,maxX,maxY}=getExploreMotionProfile({width,height,coarse});
    const frameScale=lastMotionTs?clamp((ts-lastMotionTs)/16.667,.35,3.2):1;
    lastMotionTs=ts;
    // Slow continuous drift: steer toward a gently changing heading instead of jittering.
    for(const id of ids){
      const m=motionState.get(id); if(!m)continue;
      const depth=motionDepth(id);
      const frozen=isHoverNeighborhood(id);
      const depthSpeed=(.0032 + depth*.0036)*motionScale;
      const heading=m.phase + Math.sin(t*.075 + m.phase)*.72 + Math.cos(t*.043 + m.phase*1.7)*.34;
      const targetVx=Math.cos(heading)*depthSpeed;
      const targetVy=Math.sin(heading)*depthSpeed*.82;
      const settle=frozen?.90:.018;
      m.vx += ((frozen?0:targetVx)-m.vx)*settle;
      m.vy += ((frozen?0:targetVy)-m.vy)*settle;
      const osc=getMobileOscillation({t,phase:m.phase,depth,isFocus:id===focusId,mobileProfile,frozen});
      m.oscX=osc.x;m.oscY=osc.y;
    }
    // Courteous avoidance: a broad comfort zone bends trajectories; a close zone adds firmer steering.
    for(let i=0;i<ids.length;i++)for(let j=i+1;j<ids.length;j++){
